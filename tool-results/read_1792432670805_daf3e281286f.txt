    32→    32→function toMinorSafe(v: string): number | null {
    33→    33→  const s = v.trim()
    34→    34→  if (!s) return null
    35→    35→  const n = parseFloat(s)
    36→    36→  if (!isFinite(n) || n < 0) return null
    37→    37→  return Math.round(n * 100)
    38→    38→}
    39→    39→function toMilliSafe(v: string): number | null {
    40→    40→  const s = v.trim()
    41→    41→  if (!s) return null
    42→    42→  const n = parseFloat(s)
    43→    43→  if (!isFinite(n) || n < 0) return null
    44→    44→  return Math.round(n * 1000)
    45→    45→}
