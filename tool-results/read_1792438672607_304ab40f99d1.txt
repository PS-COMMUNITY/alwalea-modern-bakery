    61→function sellFormFromSettings(s: Settings): SellForm {
    62→  return {

    66→    requireCustomerForCredit: s.requireCustomerForCredit,
    67→    receiptWidth: String(s.receiptWidth),
    68→    idleLockMinutes: String(s.idleLockMinutes),
    69→  }

   191→  async function saveSell() {
   192→    setSellError('')

   194→    if (!isFinite(pct) || pct < 0 || pct > 100) {
   195→      setSellError('نسبة الضريبة يجب أن تكون بين 0 و 100')
   196→      return
   197→    }
   198→    const idle = parseInt(sell.idleLockMinutes, 10)
   199→    if (!Number.isInteger(idle) || idle < 1 || idle > 240) {
   200→      setSellError('قفل الخمول يجب أن يكون بين 1 و 240 دقيقة')
   201→      return
   202→    }
   203→    setSellBusy(true)
   204→    try {
   205→      await api.put('/api/settings', {
   206→        taxEnabled: sell.taxEnabled,
   207→        taxRateBp: Math.round(pct * 100),
   208→        negativeStockAllowed: sell.negativeStockAllowed,
   209→        requireCustomerForCredit: sell.requireCustomerForCredit,
   210→        receiptWidth: parseInt(sell.receiptWidth, 10) === 58 ? 58 : 80,
   211→        idleLockMinutes: idle,
   212→      })
   213→      toast({ title: 'تم حفظ إعدادات البيع' })
   214→      if (onSettingsChanged) await onSettingsChanged()

   224→    setNewUserError('')

   227→      setNewUserError('اسم المستخدم: حروف إنجليزية وأرقام فقط، 3 أحرف على الأقل')

   229→    }
   230→    if (!newUser.displayName.trim()) {
   231→      setNewUserError('الاسم المعروض مطلوب')
   232→      return
   233→    }
   234→    if (newUser.password.length < 8) {
   235→      setNewUserError('كلمة المرور يجب أن تكون 8 أحرف على الأقل')
   236→      return
   237→    }
   238→    setNewUserBusy(true)
   239→    try {
   240→      await api.post('/api/users', {
   241→        username,
   242→        displayName: newUser.displayName.trim(),
   243→        role: newUser.role,
   244→        password: newUser.password,
   245→      })
   246→      toast({ title: `أُنشئ المستخدم «${username}»`, description: `الدور: ${ROLE_LABELS[newUser.role as Role] ?? newUser.role}` })

   248→      setNewUser({ username: '', displayName: '', role: 'CASHIER', password: '' })
   249→      void loadUsers()
   250→    } catch (e) {
   251→      setNewUserError(e instanceof ApiError ? e.message : 'تعذر إنشاء المستخدم')
   252→    } finally {
   253→      setNewUserBusy(false)
   254→    }
   255→  }

   302→      setNewPassword('')

   305→      setPwError(e instanceof ApiError ? e.message : 'تعذر تغيير كلمة المرور')

   328→  function onPickFile(file: File | null | undefined) {
   329→    setPreview(null)
   330→    setConfirmUnderstood(false)
   331→    setBackupError('')
   332→    setFileData(null)
   333→    setFileName('')
   334→    if (!file) return
   335→    setFileName(file.name)
   336→    const reader = new FileReader()
   337→    reader.onload = () => {
   338→      if (typeof reader.result === 'string') setFileData(reader.result)
   339→    }
   340→    reader.onerror = () => {
   341→      setBackupError('تعذر قراءة الملف')
   342→    }
   343→    reader.readAsText(file)
   344→  }

   346→  async function runPreview() {
   347→    setBackupError('')
   348→    if (!fileData) {
   349→      setBackupError('اختر ملف النسخة الاحتياطية أولًا')
   350→      return

   353→    try {
   354→      const res = await api.post<PreviewResponse>('/api/backup/preview', { data: fileData })
   355→      setPreview(res)
   356→      setConfirmUnderstood(false)
   357→    } catch (e) {
   358→      setPreview(null)
   359→      setBackupError(e instanceof ApiError ? e.message : 'تعذر فحص الملف')
   360→    } finally {
   361→      setBackupBusy(false)
   362→    }
   363→  }

   365→  async function runRestore() {
   366→    if (!fileData || !preview) return

   369→      const res = await api.post<RestoreResponse>('/api/backup/restore', { data: fileData, confirm: true })

   376→      if (onSettingsChanged) await onSettingsChanged()

   379→      setBackupError(e instanceof ApiError ? e.message : 'تعذر تنفيذ الاستعادة')
   380→    } finally {
   381→      setBackupBusy(false)
   382→    }
   383→  }
