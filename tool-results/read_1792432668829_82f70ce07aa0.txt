    32→function toMinorSafe(v: string): number | null {
    33→  const s = v.trim()
    34→  if (!s) return null
    35→  const n = parseFloat(s)
    36→  if (!isFinite(n) || n < 0) return null
    37→  return Math.round(n * 100)
    38→}
    39→function toMilliSafe(v: string): number | null {
    40→  const s = v.trim()
    41→  if (!s) return null
    42→  const n = parseFloat(s)
    43→  if (!isFinite(n) || n < 0) return null
    44→  return Math.round(n * 1000)
    45→}

   178→        cost: (prod.costMinor / 100).toString(),

   188→    }

   198→  async function submitForm() {

   201→    if (!name) {
   202→      setFormError('اسم المنتج مطلوب')

   205→    const price = toMinorSafe(form.price)

   208→      return

   210→    const unit = form.unitChoice === OTHER_UNIT ? form.unitCustom.trim() : form.unitChoice
   211→    if (!unit) {
   212→      setFormError('أدخل وحدة القياس')
   213→      return

   215→    const body: Record<string, unknown> = {
   216→      name,
   217→      code: form.code.trim() || null,
   218→      barcode: form.barcode.trim() || null,

   220→      type: form.type,
   221→      baseUnit: unit,
   222→      trackStock: form.trackStock,
   223→      priceMinor: price,
   224→      wholesalePriceMinor: toMinorSafe(form.wholesalePrice),
   225→      costMinor: toMinorSafe(form.cost) ?? 0,
   226→      minStockMilli: toMilliSafe(form.minStock) ?? 0,
   227→    }
   228→
   229→    setFormBusy(true)
   230→    try {
   231→      if (editingId) {
   232→        await api.put(`/api/products/${editingId}`, body)
   233→        toast({ title: 'تم حفظ تعديلات المنتج' })
   234→      } else {
   235→        await api.post('/api/products', body)
   236→        toast({ title: `تم إضافة المنتج «${name}»` })
   237→      }
   238→      setFormOpen(false)
   239→      void load()

   241→      setFormError(e instanceof ApiError ? e.message : 'تعذر حفظ المنتج')

   243→      setFormBusy(false)
   244→    }
   245→  }

   257→        title: 'تعذر تغيير حالة المنتج',

   262→      setToggleBusy(false)

   313→            المنتجات

   317→            التصنيفات

   490→                    {canManageCats ? <th className="p-3 text-start font-semibold">إجراءات</th> : null}
   491→                  </tr>
   492→                </thead>
   493→                <tbody>
   494→                  {categories.map(c => (
   495→                    <tr key={c.id} className="border-t transition-colors hover:bg-cream-100">
   496→                      <td className="p-3 font-semibold">{c.name}</td>
   497→                      <td className="p-3 tabular-nums">{c._count?.products ?? 0}</td>
   498→                      <td className="p-3">
   499→                        {c.active ? (
   500→                          <Badge className="bg-ok-600 text-white gap-1"><span aria-hidden>✓</span> مفعّل</Badge>
   501→                        ) : (
   502→                          <Badge variant="outline" className="gap-1"><span aria-hidden>⏸</span> معطّل</Badge>
   503→                        )}
   504→                      </td>
   505→                      {canManageCats ? (
   506→                        <td className="p-3">
   507→                          <Button type="button" variant="outline" size="sm" className="h-9" onClick={() => openCatEdit(c)} aria-label={`تعديل تصنيف ${c.name}`}>
   508→                            <Pencil className="h-4 w-4" aria-hidden />
   509→                            تعديل
   510→                          </Button>
   511→                        </td>
   512→                      ) : null}
   513→                    </tr>
   514→                  ))}
   515→                </tbody>
   516→              </table>
   517→            </TableShell>
   518→          )}
   519→        </TabsContent>
   520→      </Tabs>
   521→
   522→      {/* ===== نموذج المنتج (إضافة/تعديل) ===== */}
   523→      <Dialog open={formOpen} onOpenChange={setFormOpen}>
   524→        <DialogContent className="max-h-[90vh] max-w-2xl overflow-y-auto">
   525→          <DialogHeader>
   526→            <DialogTitle>{editingId ? 'تعديل منتج' : 'منتج جديد'}</DialogTitle>
   527→            <DialogDescription>
   528→              الأموال تُحفظ بالأغورة والكميات بالألفية تلقائيًا — أدخل القيم كما تظهر على الشاشة.
   529→            </DialogDescription>
   530→          </DialogHeader>
   531→
   532→          <div className="grid gap-3 sm:grid-cols-2">
   533→            <div className="space-y-1.5 sm:col-span-2">
   534→              <Label htmlFor="p-name">اسم المنتج *</Label>
   535→              <Input id="p-name" value={form.name} onChange={e => setForm(f => ({ ...f, name: e.target.value }))} className="h-10" autoComplete="off" />
   536→            </div>
   537→            <div className="space-y-1.5">
   538→              <Label htmlFor="p-code">الكود</Label>
   539→              <Input id="p-code" value={form.code} onChange={e => setForm(f => ({ ...f, code: e.target.value }))} className="h-10" dir="ltr" autoComplete="off" />
   540→            </div>
   541→            <div className="space-y-1.5">
   542→              <Label htmlFor="p-barcode">الباركود</Label>
   543→              <Input id="p-barcode" value={form.barcode} onChange={e => setForm(f => ({ ...f, barcode: e.target.value }))} className="h-10" dir="ltr" autoComplete="off" />
   544→            </div>
   545→            <div className="space-y-1.5">
   546→              <Label>التصنيف</Label>
   547→              <Select value={form.categoryId || '__none__'} onValueChange={v => setForm(f => ({ ...f, categoryId: v === '__none__' ? '' : v }))}>
   548→                <SelectTrigger className="h-10 w-full" aria-label="التصنيف"><SelectValue /></SelectTrigger>
   549→                <SelectContent>
   550→                  <SelectItem value="__none__">بدون تصنيف</SelectItem>
   551→                  {categories.map(c => (
   552→                    <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   553→                  ))}
   554→                </SelectContent>
   555→              </Select>
   556→            </div>
   557→            <div className="space-y-1.5">
   558→              <Label>النوع</Label>
   559→              <Select value={form.type} onValueChange={v => setForm(f => ({ ...f, type: v }))}>
   560→                <SelectTrigger className="h-10 w-full" aria-label="نوع المنتج"><SelectValue /></SelectTrigger>
   561→                <SelectContent>
   562→                  {Object.entries(PRODUCT_TYPE_LABELS).map(([k, v]) => (
   563→                    <SelectItem key={k} value={k}>{v}</SelectItem>
   564→                  ))}
   565→                </SelectContent>
   566→              </Select>
   567→            </div>
   568→            <div className="space-y-1.5">
   569→              <Label>وحدة القياس</Label>
   570→              <Select value={form.unitChoice} onValueChange={v => setForm(f => ({ ...f, unitChoice: v }))}>
   571→                <SelectTrigger className="h-10 w-full" aria-label="وحدة القياس"><SelectValue /></SelectTrigger>
   572→                <SelectContent>
   573→                  {UNIT_OPTIONS.map(u => (
   574→                    <SelectItem key={u} value={u}>{u}</SelectItem>
   575→                  ))}
   576→                  <SelectItem value={OTHER_UNIT}>أخرى…</SelectItem>
   577→                </SelectContent>
   578→              </Select>
   579→              {form.unitChoice === OTHER_UNIT ? (
   580→                <Input
   581→                  value={form.unitCustom}
   582→                  onChange={e => setForm(f => ({ ...f, unitCustom: e.target.value }))}
   583→                  placeholder="اكتب الوحدة (مثال: صينية)"
   584→                  className="h-10"
   585→                  autoComplete="off"
   586→                  aria-label="وحدة قياس مخصصة"
   587→                />
   588→              ) : null}
   589→            </div>
   590→            <div className="flex items-end gap-3 pb-1">
   591→              <div className="flex items-center gap-2">
   592→                <Switch
   593→                  id="p-track"
   594→                  checked={form.trackStock}
   595→                  onCheckedChange={v => setForm(f => ({ ...f, trackStock: v }))}
   596→                />
   597→                <Label htmlFor="p-track">تتبع المخزون</Label>
   598→              </div>
   599→            </div>
   600→            <div className="space-y-1.5">
   601→              <Label htmlFor="p-price">سعر البيع ({settings.currencySymbol}) *</Label>
   602→              <Input id="p-price" inputMode="decimal" value={form.price} onChange={e => setForm(f => ({ ...f, price: e.target.value }))} className="h-10 tabular-nums" autoComplete="off" />
   603→            </div>
   604→            <div className="space-y-1.5">
   605→              <Label htmlFor="p-wprice">سعر الجملة (اختياري)</Label>
   606→              <Input id="p-wprice" inputMode="decimal" value={form.wholesalePrice} onChange={e => setForm(f => ({ ...f, wholesalePrice: e.target.value }))} className="h-10 tabular-nums" autoComplete="off" />
   607→            </div>
   608→            <div className="space-y-1.5">
   609→              <Label htmlFor="p-cost">التكلفة ({settings.currencySymbol})</Label>
   610→              <Input id="p-cost" inputMode="decimal" value={form.cost} onChange={e => setForm(f => ({ ...f, cost: e.target.value }))} className="h-10 tabular-nums" autoComplete="off" />
   611→            </div>
   612→            <div className="space-y-1.5">
   613→              <Label htmlFor="p-minstock">الحد الأدنى للمخزون ({form.unitChoice === OTHER_UNIT ? 'الوحدة' : form.unitChoice})</Label>
   614→              <Input id="p-minstock" inputMode="decimal" value={form.minStock} onChange={e => setForm(f => ({ ...f, minStock: e.target.value }))} className="h-10 tabular-nums" autoComplete="off" />
   615→            </div>
   616→          </div>
   617→
   618→          {formError ? (
   619→            <p className="rounded-lg border border-destructive/40 bg-destructive/5 p-2 text-sm font-semibold text-destructive" role="alert">{formError}</p>
   620→          ) : null}
   621→
   622→          <DialogFooter className="gap-2">
   623→            <Button type="button" variant="outline" className="h-10" onClick={() => setFormOpen(false)} disabled={formBusy}>إلغاء</Button>
   624→            <Button type="button" className="h-10" onClick={() => void submitForm()} disabled={formBusy}>
   625→              {formBusy ? 'جارٍ الحفظ…' : editingId ? 'حفظ التعديلات' : 'إضافة المنتج'}
   626→            </Button>
   627→          </DialogFooter>
   628→        </DialogContent>
   629→      </Dialog>
   630→
   631→      {/* ===== تأكيد التعطيل/التفعيل ===== */}
   632→      <AlertDialog open={!!toggleTarget} onOpenChange={v => { if (!v) setToggleTarget(null) }}>
   633→        <AlertDialogContent>
   634→          <AlertDialogHeader>
   635→            <AlertDialogTitle>
   636→              {toggleTarget?.active ? `تعطيل «${toggleTarget?.name}»؟` : `تفعيل «${toggleTarget?.name}»؟`}
   637→            </AlertDialogTitle>
   638→            <AlertDialogDescription>
   639→              المنتج المستخدم في حركات لا يُحذف — يُعطّل فقط، ويختفي من نقطة البيع دون فقدان سجلاته.
   640→            </AlertDialogDescription>
   641→          </AlertDialogHeader>
   642→          <AlertDialogFooter>
   643→            <AlertDialogCancel className="h-10">رجوع</AlertDialogCancel>
   644→            <AlertDialogAction
   645→              className="h-10"
   646→              disabled={toggleBusy}
   647→              onClick={e => { e.preventDefault(); void toggleActive() }}
   648→            >
   649→              {toggleBusy ? 'جارٍ التنفيذ…' : 'تأكيد'}
   650→            </AlertDialogAction>
   651→          </AlertDialogFooter>
   652→        </AlertDialogContent>
   653→      </AlertDialog>
   654→
   655→      {/* ===== نموذج التصنيف ===== */}
   656→      <Dialog open={catDlgOpen} onOpenChange={setCatDlgOpen}>
   657→        <DialogContent className="max-w-sm">
   658→          <DialogHeader>
   659→            <DialogTitle>{catEditing ? 'تعديل تصنيف' : 'تصنيف جديد'}</DialogTitle>
   660→            <DialogDescription>اسم التصنيف يجب أن يكون فريدًا.</DialogDescription>
   661→          </DialogHeader>
   662→          <div className="space-y-3">
   663→            <div className="space-y-1.5">
   664→              <Label htmlFor="cat-name">اسم التصنيف *</Label>
   665→              <Input id="cat-name" value={catForm.name} onChange={e => setCatForm(f => ({ ...f, name: e.target.value }))} className="h-10" autoComplete="off" />
   666→            </div>
   667→            {catEditing ? (
   668→              <div className="flex items-center gap-2">
   669→                <Switch id="cat-active" checked={catForm.active} onCheckedChange={v => setCatForm(f => ({ ...f, active: v }))} />
   670→                <Label htmlFor="cat-active">مفعّل</Label>
   671→              </div>
   672→            ) : null}
   673→          </div>
   674→          <DialogFooter className="gap-2">
   675→            <Button type="button" variant="outline" className="h-10" onClick={() => setCatDlgOpen(false)} disabled={catBusy}>إلغاء</Button>
   676→            <Button type="button" className="h-10" onClick={() => void submitCat()} disabled={catBusy || !catForm.name.trim()}>
   677→              {catBusy ? 'جارٍ الحفظ…' : 'حفظ'}
   678→            </Button>
   679→          </DialogFooter>
   680→        </DialogContent>
   681→      </Dialog>
   682→    </div>
   683→  )
   684→}
