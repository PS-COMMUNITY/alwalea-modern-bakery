    88→/** تاريخ اليوم بصيغة YYYY-MM-DD */
    89→function today(): string {
    90→  const d = new Date()
    91→  const pad = (n: number): string => String(n).padStart(2, '0')
    92→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
    93→}

    95→export default function DistributionScreen(props: { user: SessionUser }): ReactNode {
    96→  const [reps, setReps] = useState<RepRow[] | null>(null)
    97→  const [routes, setRoutes] = useState<RouteRow[] | null>(null)
    98→  const [trips, setTrips] = useState<TripListRow[] | null>(null)
    99→  const [products, setProducts] = useState<ProductLite[]>([])
   100→  const [customers, setCustomers] = useState<CustomerLite[]>([])
   101→  const [statusFilter, setStatusFilter] = useState('')
   102→  const [error, setError] = useState('')
   103→  const [notice, setNotice] = useState('')
   104→  const [busy, setBusy] = useState(false)
   105→
   106→  // نوافذ المندوبين والخطوط والجولات
   107→  const [repForm, setRepForm] = useState<{ name: string; phone: string } | null>(null)
   108→  const [routeForm, setRouteForm] = useState<{ name: string; note: string } | null>(null)
   109→  const [tripForm, setTripForm] = useState<{ repId: string; routeId: string; date: string; notes: string } | null>(null)
   110→
   111→  // تفاصيل الجولة + النماذج داخلها
   112→  const [detail, setDetail] = useState<TripDetail | null>(null)
   113→  const [loadForm, setLoadForm] = useState<{ productId: string; qty: string }>({ productId: '', qty: '' })
   114→  const [saleForm, setSaleForm] = useState<{ productId: string; qty: string; price: string; payType: 'CASH' | 'CREDIT'; customerId: string }>({
   115→    productId: '',
   116→    qty: '',
   117→    price: '',
   118→    payType: 'CASH',
   119→    customerId: ''
   120→  })
   121→  const [returnForm, setReturnForm] = useState<{ productId: string; qty: string; condition: 'GOOD' | 'DAMAGED'; note: string }>({
   122→    productId: '',
   123→    qty: '',
   124→    condition: 'GOOD',
   125→    note: ''
   126→  })
   127→
   128→  // التسوية والإلغاء
   129→  const [settleFor, setSettleFor] = useState<TripDetail | null>(null)
   130→  const [settleCounted, setSettleCounted] = useState('')
   131→  const [settleNotes, setSettleNotes] = useState('')
   132→  const [cancelFor, setCancelFor] = useState<TripListRow | null>(null)
   133→  const [cancelReason, setCancelReason] = useState('')
   134→
   135→  const canManage = props.user.perms.includes('distribution.manage')
   136→
   137→  const load = useCallback(async (): Promise<void> => {
   138→    setError('')
   139→    try {
   140→      const tripsPayload: Record<string, unknown> = {}
   141→      if (statusFilter) tripsPayload.status = statusFilter
   142→      const [r, rt, t, p, c] = await Promise.all([
   143→        call<RepRow[]>('distribution.reps.list', { includeInactive: true }),
   144→        call<RouteRow[]>('distribution.routes.list', { includeInactive: true }),
   145→        call<TripListRow[]>('distribution.trips.list', tripsPayload),
   146→        call<ProductLite[]>('products.list', { includeInactive: true }),
   147→        call<CustomerLite[]>('customers.list', { includeInactive: true })
   148→      ])
   149→      setReps(r)
   150→      setRoutes(rt)
   151→      setTrips(t)
   152→      setProducts(p)
   153→      setCustomers(c)
   154→    } catch (e) {
   155→      setError(e instanceof Error ? e.message : String(e))
   156→    }
   157→  }, [statusFilter])
   158→
   159→  useEffect(() => {
   160→    void load()
   161→  }, [load])
   162→
   163→  const openDetail = useCallback(async (id: number): Promise<void> => {
   164→    setError('')
   165→    try {
   166→      const d = await call<TripDetail>('distribution.trips.get', { id })
   167→      setDetail(d)
   168→      setLoadForm({ productId: '', qty: '' })
   169→      setSaleForm({ productId: '', qty: '', price: '', payType: 'CASH', customerId: '' })
   170→      setReturnForm({ productId: '', qty: '', condition: 'GOOD', note: '' })
   171→    } catch (e) {
   172→      setError(e instanceof Error ? e.message : String(e))
   173→    }
   174→  }, [])
   175→
   176→  const refreshDetail = async (id: number): Promise<void> => {
   177→    const d = await call<TripDetail>('distribution.trips.get', { id })
   178→    setDetail(d)
   179→  }
   180→
   181→  // ---------- مندوب / خط / جولة ----------
   182→  const saveRep = async (): Promise<void> => {
   183→    if (!repForm) return
   184→    if (!repForm.name.trim()) {
   185→      setError('اسم المندوب مطلوب')
   186→      return
   187→    }
   188→    setBusy(true)
   189→    setError('')
   190→    try {
   191→      await call('distribution.reps.create', { name: repForm.name.trim(), phone: repForm.phone.trim() || null })
   192→      setRepForm(null)
   193→      setNotice('تمت إضافة المندوب')
   194→      await load()
   195→    } catch (e) {
   196→      setError(e instanceof Error ? e.message : String(e))
   197→    } finally {
   198→      setBusy(false)
   199→    }
   200→  }
   201→
   202→  const saveRoute = async (): Promise<void> => {
   203→    if (!routeForm) return
   204→    if (!routeForm.name.trim()) {
   205→      setError('اسم الخط مطلوب')
   206→      return
   207→    }
   208→    setBusy(true)
   209→    setError('')
   210→    try {
   211→      await call('distribution.routes.create', { name: routeForm.name.trim(), note: routeForm.note.trim() || null })
   212→      setRouteForm(null)
   213→      setNotice('تمت إضافة الخط')
   214→      await load()
   215→    } catch (e) {
   216→      setError(e instanceof Error ? e.message : String(e))
   217→    } finally {
   218→      setBusy(false)
   219→    }
   220→  }
   221→
   222→  const saveTrip = async (): Promise<void> => {
   223→    if (!tripForm) return
   224→    setError('')
   225→    const repId = Number(tripForm.repId)
   226→    if (!repId) {
   227→      setError('اختر المندوب')
   228→      return
   229→    }
   230→    setBusy(true)
   231→    try {
   232→      const payload: Record<string, unknown> = { repId, notes: tripForm.notes.trim() || null }
   233→      if (tripForm.routeId) payload.routeId = Number(tripForm.routeId)
   234→      if (tripForm.date) payload.date = tripForm.date
   235→      const res = await call<TripDetail>('distribution.trips.create', payload)
   236→      setTripForm(null)
   237→      setNotice(`تم فتح الجولة ${res.trip_no} — حمّل المنتجات ثم سجّل البيع والمرتجعات`)
   238→      await load()
   239→      await openDetail(res.id)
   240→    } catch (e) {
   241→      setError(e instanceof Error ? e.message : String(e))
   242→    } finally {
   243→      setBusy(false)
   244→    }
   245→  }
   246→
   247→  // ---------- عمليات داخل الجولة ----------
   248→  const submitLoad = async (): Promise<void> => {
   249→    if (!detail) return
   250→    setError('')
   251→    const productId = Number(loadForm.productId)
   252→    const qtyMilli = toMilli(loadForm.qty)
   253→    if (!productId) {
   254→      setError('اختر المنتج للتحميل')
   255→      return
   256→    }
   257→    if (qtyMilli === null || qtyMilli <= 0) {
   258→      setError('كمية التحميل يجب أن تكون أكبر من صفر')
   259→      return
   260→    }
   261→    setBusy(true)
   262→    try {
   263→      await call('distribution.trips.load', { tripId: detail.id, productId, qtyMilli })
   264→      setLoadForm({ productId: '', qty: '' })
   265→      setNotice('تم تحميل المنتج على عربة الجولة')
   266→      await refreshDetail(detail.id)
   267→      await load()
   268→    } catch (e) {
   269→      setError(e instanceof Error ? e.message : String(e))
   270→    } finally {
   271→      setBusy(false)
   272→    }
   273→  }
   274→
   275→  const submitSale = async (): Promise<void> => {
   276→    if (!detail) return
   277→    setError('')
   278→    const productId = Number(saleForm.productId)
   279→    const qtyMilli = toMilli(saleForm.qty)
   280→    const unitPriceMinor = toMinor(saleForm.price)
   281→    if (!productId) {
   282→      setError('اختر المنتج المبيع')
   283→      return
   284→    }
   285→    if (qtyMilli === null || qtyMilli <= 0) {
   286→      setError('كمية البيع يجب أن تكون أكبر من صفر')
   287→      return
   288→    }
   289→    if (unitPriceMinor === null) {
   290→      setError('سعر الوحدة غير صحيح')
   291→      return
   292→    }
   293→    if (saleForm.payType === 'CREDIT' && !saleForm.customerId) {
   294→      setError('البيع الآجل يتطلب تحديد العميل')
   295→      return
   296→    }
   297→    setBusy(true)
   298→    try {
   299→      await call('distribution.trips.sale', {
   300→        tripId: detail.id,
   301→        items: [
   302→          {
   303→            productId,
   304→            qtyMilli,
   305→            unitPriceMinor,
   306→            paymentType: saleForm.payType,
   307→            customerId: saleForm.payType === 'CREDIT' ? Number(saleForm.customerId) : null
   308→          }
   309→        ]
   310→      })
   311→      setSaleForm({ productId: '', qty: '', price: '', payType: saleForm.payType, customerId: '' })
   312→      setNotice('سُجّل البيع على الجولة — يُرحّل عند التسوية')
   313→      await refreshDetail(detail.id)
   314→      await load()
   315→    } catch (e) {
   316→      setError(e instanceof Error ? e.message : String(e))
   317→    } finally {
   318→      setBusy(false)
   319→    }
   320→  }
   321→
   322→  const submitReturn = async (): Promise<void> => {
   323→    if (!detail) return
   324→    setError('')
   325→    const productId = Number(returnForm.productId)
   326→    const qtyMilli = toMilli(returnForm.qty)
   327→    if (!productId) {
   328→      setError('اختر منتج المرتجع')
   329→      return
   330→    }
   331→    if (qtyMilli === null || qtyMilli <= 0) {
   332→      setError('كمية المرتجع يجب أن تكون أكبر من صفر')
   333→      return
   334→    }
   335→    setBusy(true)
   336→    try {
   337→      await call('distribution.trips.return', {
   338→        tripId: detail.id,
   339→        items: [{ productId, qtyMilli, condition: returnForm.condition, note: returnForm.note.trim() || null }]
   340→      })
   341→      setReturnForm({ productId: '', qty: '', condition: 'GOOD', note: '' })
   342→      setNotice(returnForm.condition === 'GOOD' ? 'أُعيدت الكمية للمخزون (سليمة)' : 'وُثق المرتجع التالف كهالك بعد إعادته للمخزون')
   343→      await refreshDetail(detail.id)
   344→      await load()
   345→    } catch (e) {
   346→      setError(e instanceof Error ? e.message : String(e))
   347→    } finally {
   348→      setBusy(false)
   349→    }
   350→  }
   351→
   352→  const submitSettle = async (): Promise<void> => {
   353→    if (!settleFor) return
   354→    setError('')
   355→    const countedMinor = settleCounted.trim() === '' ? null : toMinor(settleCounted)
   356→    if (settleCounted.trim() !== '' && countedMinor === null) {
   357→      setError('النقد المجرود غير صحيح')
   358→      return
   359→    }
   360→    setBusy(true)
   361→    try {
   362→      const payload: Record<string, unknown> = { tripId: settleFor.id }
   363→      if (countedMinor !== null) payload.countedCashMinor = countedMinor
   364→      if (settleNotes.trim()) payload.notes = settleNotes.trim()
   365→      await call('distribution.trips.settle', payload)
   366→      const no = settleFor.trip_no
   367→      setSettleFor(null)
   368→      setSettleCounted('')
   369→      setSettleNotes('')
   370→      setNotice(`سُوّت الجولة ${no}: رُحّلت فواتير البيع ودخل النقد للصندوق وقُيّد الآجل على دفتر العملاء`)
   371→      await load()
   372→      await openDetail(settleFor.id)
   373→    } catch (e) {
   374→      setError(e instanceof Error ? e.message : String(e))
   375→    } finally {
   376→      setBusy(false)
   377→    }
   378→  }
   379→
   380→  const submitCancel = async (): Promise<void> => {
   381→    if (!cancelFor) return
   382→    if (cancelReason.trim().length < 3) {
   383→      setError('اكتب سبب الإلغاء (3 أحرف على الأقل)')
   384→      return
   385→    }
   386→    setBusy(true)
   387→    setError('')
   388→    try {
   389→      await call('distribution.trips.cancel', { id: cancelFor.id, reason: cancelReason.trim() })
   390→      setCancelFor(null)
   391→      setCancelReason('')
   392→      setNotice('أُلغيت الجولة ورُد تحميلها للمخزون')
   393→      await load()
   394→    } catch (e) {
   395→      setError(e instanceof Error ? e.message : String(e))
   396→    } finally {
   397→      setBusy(false)
   398→    }
   399→  }
   400→
   401→  const onSaleProductChange = (pid: string): void => {
   402→    const p = products.find((x) => String(x.id) === pid)
   403→    setSaleForm((f) => ({ ...f, productId: pid, price: p ? (p.price_minor / 100).toFixed(2) : f.price }))
   404→  }
   405→
   406→  return (
   407→    <div>
   408→      <div style={{ display: 'flex', gap: 10, alignItems: 'center', flexWrap: 'wrap', marginBottom: 14 }}>
   409→        <h1 style={{ marginRight: 'auto' }}>التوزيع (الجولات)</h1>
   410→        {canManage ? (
   411→          <>
   412→            <button className="btn ghost" onClick={() => setRepForm({ name: '', phone: '' })}>
   413→              ＋ مندوب
   414→            </button>
   415→            <button className="btn ghost" onClick={() => setRouteForm({ name: '', note: '' })}>
   416→              ＋ خط توزيع
   417→            </button>
   418→            <button className="btn primary" onClick={() => setTripForm({ repId: '', routeId: '', date: today(), notes: '' })}>
   419→              ＋ جولة جديدة
   420→            </button>
   421→          </>
   422→        ) : null}
   423→      </div>
   424→
   425→      {error ? <Banner kind="error" text={error} /> : null}
   426→      {notice ? <Banner kind="ok" text={notice} /> : null}
   427→
   428→      {/* ================= المندوبون والخطوط ================= */}
   429→      <div className="grid-cards" style={{ marginBottom: 22, gridTemplateColumns: 'repeat(auto-fit, minmax(300px, 1fr))' }}>
   430→        <div className="card" style={{ padding: 14 }}>
   431→          <h3>🛵 المندوبون</h3>
   432→          {reps === null ? (
   433→            <EmptyState icon="⏳" text="جارٍ التحميل…" />
   434→          ) : reps.length === 0 ? (
   435→            <EmptyState icon="🛵" text="لا مندوبين بعد." />
   436→          ) : (
   437→            <table className="data">
   438→              <thead>
   439→                <tr>
   440→                  <th>الاسم</th>
   441→                  <th>الهاتف</th>
   442→                  <th>الحالة</th>
   443→                </tr>
   444→              </thead>
   445→              <tbody>
   446→                {reps.map((r) => (
   447→                  <tr key={r.id}>
   448→                    <td>{r.name}</td>
   449→                    <td dir="ltr">{r.phone ?? '—'}</td>
   450→                    <td>{r.active === 1 ? <Badge kind="ok" text="نشط" /> : <Badge kind="muted" text="معطّل" />}</td>
   451→                  </tr>
   452→                ))}
   453→              </tbody>
   454→            </table>
   455→          )}
   456→        </div>
   457→        <div className="card" style={{ padding: 14 }}>
   458→          <h3>🗺️ خطوط التوزيع</h3>
   459→          {routes === null ? (
   460→            <EmptyState icon="⏳" text="جارٍ التحميل…" />
   461→          ) : routes.length === 0 ? (
   462→            <EmptyState icon="🗺️" text="لا خطوط بعد." />
   463→          ) : (
   464→            <table className="data">
   465→              <thead>
   466→                <tr>
   467→                  <th>الخط</th>
   468→                  <th>ملاحظة</th>
   469→                  <th>الحالة</th>
   470→                </tr>
   471→              </thead>
   472→              <tbody>
   473→                {routes.map((r) => (
   474→                  <tr key={r.id}>
   475→                    <td>{r.name}</td>
   476→                    <td>{r.note ?? '—'}</td>
   477→                    <td>{r.active === 1 ? <Badge kind="ok" text="نشط" /> : <Badge kind="muted" text="معطّل" />}</td>
   478→                  </tr>
   479→                ))}
   480→              </tbody>
   481→            </table>
   482→          )}
   483→        </div>
   484→      </div>
   485→
   486→      {/* ================= الجولات ================= */}
   487→      <div style={{ display: 'flex', gap: 10, alignItems: 'center', flexWrap: 'wrap', marginBottom: 10 }}>
   488→        <h2 style={{ marginRight: 'auto' }}>🚚 الجولات</h2>
   489→        <div className="field" style={{ marginBottom: 0 }}>
   490→          <label>الحالة</label>
   491→          <select value={statusFilter} onChange={(e) => setStatusFilter(e.target.value)}>
   492→            <option value="">الكل</option>
   493→            <option value="OPEN">مفتوحة</option>
   494→            <option value="SETTLED">مسواة</option>
   495→            <option value="CANCELLED">ملغاة</option>
   496→          </select>
   497→        </div>
   498→      </div>
   499→      {trips === null ? (
   500→        <EmptyState icon="⏳" text="جارٍ التحميل…" />
   501→      ) : trips.length === 0 ? (
   502→        <EmptyState icon="🚚" text="لا جولات مطابقة — افتح جولة: اختر مندوبًا وحمّل المنتجات على العربة." />
   503→      ) : (
   504→        <table className="data">
   505→          <thead>
   506→            <tr>
   507→              <th>الجولة</th>
   508→              <th>التاريخ</th>
   509→              <th>المندوب</th>
   510→              <th>الخط</th>
   511→              <th className="num">المحمّل</th>
   512→              <th>الحالة</th>
   513→              <th>سُوّت في</th>
   514→              <th></th>
   515→            </tr>
   516→          </thead>
   517→          <tbody>
   518→            {trips.map((t) => (
   519→              <tr key={t.id}>
   520→                <td>
   521→                  <strong>{t.trip_no}</strong>
